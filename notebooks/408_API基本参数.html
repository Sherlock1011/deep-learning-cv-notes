03.流式传输.py

代码
# 导入 OpenAI 模块，用于与 DeepSeek 的 API 通信
from openai import OpenAI  
# 导入 os 模块，用于访问环境变量
import os  

# 从系统环境变量中读取 API 密钥
api_key = os.environ['DEEPSEEK_API_KEY']  

# 创建 OpenAI 客户端对象，设置 base_url 为 DeepSeek 网关地址，附带密钥
client = OpenAI(
    base_url='https://api.deepseek.com',  # DeepSeek API 网关地址
    api_key=api_key  # 你的 API 密钥，用于身份验证
)

# 使用 chat 接口创建一个对话请求，并启用流式传输
completion = client.chat.completions.create(
    model='deepseek-chat',  # 指定使用的模型（deepseek 提供的 chat 模型）
    messages=[  # 定义对话消息列表
        {
            'role': 'system',  # 系统提示（用于设定整体风格或限制）
            'content': "所有的回答不到得少于100个字"  # 告诉模型：回答不少于100个字
        },
        {
            'role': 'user',  # 用户消息
            'content': "你是谁"  # 用户提问
        }
    ],
    stream=True,  # 启用流式传输，返回的是一个生成器，可逐步读取内容
)

# 遍历返回的流式结果，每次获取一个响应片段
for chunk in completion:
    # 打印每个响应片段中的新增内容（delta.content），不换行
    print(chunk.choices[0].delta.content, end='')
